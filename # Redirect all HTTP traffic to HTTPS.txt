# Redirect all HTTP traffic to HTTPS

server {

    listen 80;

    server_name ainewsinsights.hyperthink.io;

    # Redirect to HTTPS

    return 301 https://$host$request_uri;

}

# HTTPS server block

server {

    listen 443 ssl;
    server_name ainewsinsights.hyperthink.io www.ainewsinsights.hyperthink.io;

    # SSL configuration
    ssl_certificate /home/khadeer/certificates/newapi-cerfiticates/certs/ca_bundle-chint.crt;
    ssl_certificate_key /home/khadeer/certificates/newapi-cerfiticates/certs/private.key;

    # Logs for debugging and monitoring
    access_log /var/log/nginx/nginx.vhost.access.log;
    error_log /var/log/nginx/nginx.vhost.error.log;
    client_max_body_size 5M;

    # Proxy settings

    location / {

        proxy_pass http://localhost:5001; # Your backend application
        proxy_redirect off;
        proxy_set_header HOST $host;
        proxy_set_header Referer $http_referer;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Port 443; # Reflect HTTPS port
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_connect_timeout 300;
        proxy_set_header Connection "";
        chunked_transfer_encoding off;

    }

}
